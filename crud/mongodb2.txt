//Array operator

db.collection('users').find({ age: { $in: [18, 25, 30] } });
( Finds users whose age is either 18, 25, or 30)

db.collection('users').find({ age: { $nin: [18, 25, 30] } });
(Finds users whose age is neither 18, 25, nor 30)

db.collection('users').find({ hobbies: { $all: ['reading', 'coding'] } });
(Finds users who have both "reading" and "coding" in their hobbies array)

db.collection('orders').find({ items: { $elemMatch: { price: { $gt: 100 } } } });
(Finds orders where at least one item has a price greater than 100)

db.collection('users').find({ hobbies: { $size: 3 } });
(Finds users who have exactly 3 hobbies)

//string operator

db.collection('users').find({ name: { $regex: /^A/ } });
(Finds users whose name starts with "A")

db.collection('users').find({ name: { $regex: /^a/, $options: 'i' } });
( Finds users whose name starts with "a" (case-insensitive))
